Auszug aus der Ablaufsteuerung des Waffle-Pack-Bestückers (SRT Resistor Technology)

WafflePack_Master_Steuerung.cpp 
// PROJEKT: Waffle-Pack-Bestücker für SMD-Präzisionswiderstände
// HARDWARE: Arduino Mega 2560 R3 + 2x PI C-663.12 + SMC Pneumatik + Omron Laser
// AUTOR: Franck Adrien Mba (SRT Resistor Technology GmbH / FAU Mechatronik)

#include 
#define SerialX Serial1  // X-Achse: PI Mercury C-663.12 (UART1: TX18/RX19)
#define SerialY Serial2  // Y-Achse: PI Mercury C-663.12 (UART2: TX16/RX17)

const int PIN_OMRON_SENSOR = 22; // Omron E3C-LDA11 Laser-Präsenzsensor
const int PIN_VENTIL_SAUGEN = 24; // SMC SY114-5LZ Vakuumsauger
const int PIN_VENTIL_HUB    = 26; // SMC VQD1151 Pneumatik-Hubzylinder

void holeUndPlatziereBauteil(float zielX_mm, float zielY_mm) {
    // 1. Zur Übergabeposition fahren (Abholpunkt Schüttgut / X = 0.00 mm)
    SerialX.print("MOV 1 0.000\n");
    warteBisStillstand(SerialX);

    // 2. Hubzylinder senken & Vakuumsauger aktivieren
    digitalWrite(PIN_VENTIL_HUB, HIGH);
    delay(250);
    digitalWrite(PIN_VENTIL_SAUGEN, HIGH);
    delay(400);
    digitalWrite(PIN_VENTIL_HUB, LOW);  // Hubzylinder wieder anheben
    delay(250);

    // 3. Optische Teileprüfung über Omron Laser-Sensor
    if (digitalRead(PIN_OMRON_SENSOR) == HIGH) {
        // 4. Closed-Loop-Fahrt zum berechneten Matrix-Nest des Waffle Packs
        SerialX.print("MOV 1 " + String(zielX_mm, 3) + "\n");
        SerialY.print("MOV 1 " + String(zielY_mm, 3) + "\n");
        warteBisStillstand(SerialX);
        warteBisStillstand(SerialY);

        // 5. Bauteil präzise im Nest ablegen
        digitalWrite(PIN_VENTIL_HUB, HIGH);
        delay(250);
        digitalWrite(PIN_VENTIL_SAUGEN, LOW); // Vakuum abschalten
        delay(300);
        digitalWrite(PIN_VENTIL_HUB, LOW);
        delay(200);
    }
}
